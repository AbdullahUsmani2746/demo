CREATE OR REPLACE FUNCTION public.auto_assign_numbers()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
    seq_name TEXT;
BEGIN
    -- ---------- GRN_NO ----------
    IF NEW.tran_code = 4 THEN
        NEW.grn_no := nextval('grn_no_seq');
    END IF;

    -- ---------- DISPATCH_NO ----------
    IF NEW.tran_code = 6 THEN
        NEW.dispatch_no := nextval('dispatch_no_seq');
    END IF;

    -- ---------- VR_NO ----------
    IF NEW.tran_code IN (1,2) THEN
        -- rmk3 carries the cash/bank flag (set by the app)
        seq_name := CASE
                        WHEN NEW.tran_code = 1 AND NEW.rmk3 = 'cash' THEN 'vr_no_seq_1_cash'
                        WHEN NEW.tran_code = 1 AND NEW.rmk3 = 'bank' THEN 'vr_no_seq_1_bank'
                        WHEN NEW.tran_code = 2 AND NEW.rmk3 = 'cash' THEN 'vr_no_seq_2_cash'
                        WHEN NEW.tran_code = 2 AND NEW.rmk3 = 'bank' THEN 'vr_no_seq_2_bank'
                    END;
    ELSE
        -- For other tran_codes, get sequence from helper table
        SELECT sequence_name
        INTO seq_name
        FROM tran_code_sequences
        WHERE tran_code = NEW.tran_code;
    END IF;

    -- Assign VR number if a sequence exists
    IF seq_name IS NOT NULL THEN
        NEW.vr_no := nextval(seq_name);
    END IF;

    RETURN NEW;
END;
$function$;

-- Drop the broken functions
DROP FUNCTION IF EXISTS get_next_itcd_by_ic_id(integer) CASCADE;

-- Recreate with correct table name "Item" (case-sensitive)
CREATE OR REPLACE FUNCTION get_next_itcd_by_ic_id(ic_id_param INTEGER)
RETURNS INTEGER AS $$
DECLARE
    next_seq INTEGER;
    new_itcd INTEGER;
    last_itcd_in_category INTEGER;
BEGIN
    -- Check the LAST ITCD number in this category - use "Item" with quotes
    SELECT MAX(itcd) INTO last_itcd_in_category 
    FROM "Item" 
    WHERE ic_id = ic_id_param;
    
    -- Determine next sequence
    IF last_itcd_in_category IS NOT NULL THEN
        -- Continue from last ITCD + 1
        next_seq := (last_itcd_in_category % 100000) + 1;
    ELSE
        -- Start from sequence 1
        next_seq := 1;
    END IF;
    
    -- Generate new ITCD (ic_id * 100000 + sequence)
    new_itcd := (ic_id_param * 100000) + next_seq;
    
    RETURN new_itcd;
END;
$$ LANGUAGE plpgsql;

-- Create the trigger function
CREATE OR REPLACE FUNCTION set_item_itcd_from_ic_id()
RETURNS TRIGGER AS $$
BEGIN
    -- ALWAYS generate new itcd, regardless of what was provided
    NEW.itcd := get_next_itcd_by_ic_id(NEW.ic_id);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Create the trigger on the correct table "Item"
DROP TRIGGER IF EXISTS trigger_set_item_itcd_from_ic_id ON "Item";
CREATE TRIGGER trigger_set_item_itcd_from_ic_id
    BEFORE INSERT ON "Item"
    FOR EACH ROW
    EXECUTE FUNCTION set_item_itcd_from_ic_id();

    region_param   VARCHAR,
    category_param INTEGER
)
RETURNS INTEGER AS $$
DECLARE
    next_no INTEGER;
BEGIN
    UPDATE "order_region_sequences"
    SET last_no = last_no + 1
    WHERE region   = UPPER(region_param)
      AND category = category_param
    RETURNING last_no INTO next_no;

    IF next_no IS NULL THEN
        RAISE EXCEPTION 'No sequence found for region: % category: %',
            region_param, category_param;
    END IF;

    RETURN next_no;
END;
$$ LANGUAGE plpgsql;
